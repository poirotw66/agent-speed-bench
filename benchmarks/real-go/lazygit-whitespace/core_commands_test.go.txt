package git_commands

import "testing"

func TestASBCoreTrailerSpacing(t *testing.T) {
    for _, body := range []string{"Text\n", "Text\n\n\n"} {
        got:=AddCoAuthorToDescription(body,"Tester <test@example.org>")
        want:="Text\n\nCo-authored-by: Tester <test@example.org>"
        if got!=want {t.Fatalf("got %q, want %q",got,want)}
    }
}
